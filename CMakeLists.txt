cmake_minimum_required(VERSION 3.21)
project(traymd-omarchy VERSION 0.7.3 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.9 REQUIRED COMPONENTS Core Gui Qml Quick)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)

add_library(traymd_core STATIC backend/notestore.cpp backend/notedocument.cpp backend/noteeditor.cpp backend/highlighter.cpp backend/code_highlight.c backend/theme.cpp backend/shortcut.cpp backend/keybindings.cpp)
set_target_properties(traymd_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(traymd_core PUBLIC backend)
target_link_libraries(traymd_core PUBLIC Qt6::Quick Qt6::Qml PRIVATE PkgConfig::GLIB)
target_compile_options(traymd_core PRIVATE -Wall -Wextra -Wpedantic)

add_library(traymdbackend MODULE backend/plugin.cpp)
target_link_libraries(traymdbackend PRIVATE traymd_core)
set_target_properties(traymdbackend PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugin/TrayMDBackend")
foreach(file manifest.json BarWidget.qml NotesWidget.qml NotesPanel.qml NotesPopup.qml PopupSettings.qml NoteButton.qml EditorMenu.qml EditorMenuItem.qml LICENSE)
  configure_file("${file}" "plugin/${file}" COPYONLY)
endforeach()
# Quickshell serves local QML through a virtual filesystem. Native libraries
# need a real filesystem path, resolved for the build and again on installation.
set(TRAYMD_BACKEND_PATH "${CMAKE_BINARY_DIR}/plugin/TrayMDBackend")
configure_file(backend/qmldir plugin/TrayMDBackend/qmldir @ONLY)

include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME install-helpers COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/install_test.py")
  add_executable(traymd_tests tests/backend_test.cpp)
  target_link_libraries(traymd_tests PRIVATE traymd_core Qt6::Test)
  add_test(NAME backend COMMAND traymd_tests)
  set_tests_properties(backend PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=")
  find_program(QUICKSHELL_EXECUTABLE quickshell)
  if(QUICKSHELL_EXECUTABLE AND EXISTS "/usr/share/omarchy/shell/Ui/KeyboardPanel.qml")
    add_test(NAME ui COMMAND bash "${CMAKE_SOURCE_DIR}/scripts/test-ui.sh")
    set_tests_properties(ui PROPERTIES ENVIRONMENT "TRAYMD_BUILD_DIR=${CMAKE_BINARY_DIR}")
    find_program(QMLLINT_EXECUTABLE qmllint HINTS /usr/lib/qt6/bin)
    if(QMLLINT_EXECUTABLE)
      add_test(NAME popup-types COMMAND bash "${CMAKE_SOURCE_DIR}/scripts/test-popup.sh" "${QMLLINT_EXECUTABLE}" "${CMAKE_BINARY_DIR}")
    endif()
  endif()
endif()
